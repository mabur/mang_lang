result@{

Intersection = {
    distance = Number
    position = Vector3
    normal = Vector3
    color = Vector3
}

default_intersection = <Intersection>{
    distance = inf
    position = (0 0 0)
    normal = (0 0 0)
    color = (0 0 0)
}

Light = {
    direction = Vector3
    color = Vector3
}

Sphere = {
    position = Vector3
    squared_radius = Number
    color = Vector3
}

Lights = [Light]
Spheres = [Sphere]

World = {
    spheres = Spheres
    lights = Lights
    atmosphere_color = Vector3
}

muls = in (<Number>scalar <Vector3>vector) out
    <Vector3>mul3!((scalar scalar scalar) vector)

R = 100000.0
MAX_C = 1.0
MIN_C = 0.1

world = <World>{
    spheres = [
        {position=(-2 0 6) squared_radius=1 color=(MAX_C MAX_C MIN_C)}
        {position=(0 0 5) squared_radius=1 color=(MAX_C MIN_C MIN_C)}
        {position=(2 0 4) squared_radius=1 color=(mul!(2 MIN_C) mul!(4 MIN_C) MAX_C)}
        {position=(0 add!(1 R) 0) squared_radius=mul!(R R) color=(MIN_C MAX_C MIN_C)}
        {position=(0 sub!(-1 R) 0) squared_radius=mul!(R R) color=(MAX_C MAX_C MAX_C)}
    ]
    lights = [
        {direction=(+1 +1 +2) color=muls!(0.4 (1 0.8 0.5))}
        {direction=(-1 -1 -2) color=muls!(0.4 (0.5 0.5 1))}
    ]
    atmosphere_color = muls!(0.3 (0.5 0.5 1))
}

normalize3 = in <Vector3>vector out <Vector3>
    muls!(div!(1 norm3!vector) vector)

findSingleIntersection = in (<Vector3>start <Vector3>direction <Sphere>sphere) out <Intersection>intersection@{
    intersection = default_intersection
    offset = sub3!(position@sphere start)
    c = dot3!(direction offset)
    is_backwards = less?(c 0.0)
    if is_backwards
        return
    end
    discriminant = add!(sub!(mul!(c c) squared_norm3!offset) squared_radius@sphere)
    is_outside = less?(discriminant 0.0)
    if is_outside
        return
    end
    intersection = {
        distance = sub!(c sqrt!discriminant)
        position = add3!(start muls!(distance direction))
        normal = normalize3!sub3!(position position@sphere)
        color = color@sphere
    }
}

findIntersection = in (<Vector3>start <Vector3>direction <Spheres>spheres) out <Intersection>i1@{
    i1 = default_intersection
    for sphere in spheres
        i2 = findSingleIntersection!(start direction sphere)
        i1 = if less?(distance@i2 distance@i1) then i2 else i1
    end
}

shadeSingleLight = in (<Intersection>intersection <Light>light) out result@{
    geometry = max!(neg!dot3!(direction@light normal@intersection) 0.0)
    result = muls!(geometry mul3!(color@intersection color@light))
}

shadeAtmosphere = in (<Intersection>intersection <Vector3>atmosphere_color) out 
    <Vector3>muls!(sqrt!get2!position@intersection atmosphere_color)

shade = in (<Intersection>intersection world) out <Vector3>color@{
    color = (1 1 1)
    is_infinite = equal?(inf distance@intersection)
    if is_infinite
        return
    end
    color = shadeAtmosphere!(intersection atmosphere_color@world)
    for light in lights@world
        color = add3!(color shadeSingleLight!(intersection light))
    end
}

colorU8fromF64 = in <Number>c out <Number>round!min!(mul!(255 c) 255)

writePixel = in (<String>image <Number>x <Number>y <Number>width <Number>height <World>world) out <String>image_out@{
    start = (0 0 0)
    xd = sub!(x div!(width 2))
    yd = sub!(y div!(height 2))
    zd = div!(height 2)
    direction = normalize3!(xd yd zd)
    intersection = findIntersection!(start direction spheres@world)
    color = shade!(intersection world)

    image_out = image
    image_out ++= serialize_natural_number!colorU8fromF64!color!0
    image_out ++= " "
    image_out ++= serialize_natural_number!colorU8fromF64!color!1
    image_out ++= " "
    image_out ++= serialize_natural_number!colorU8fromF64!color!2
    image_out ++= " "
}

writeImage = in <World>world out <String>image@{
    WIDTH = 320
    HEIGHT = 240
    image = ""
    image ++= "P3"
    image += newline
    image ++= serialize_natural_number!WIDTH
    image += newline
    image ++= serialize_natural_number!HEIGHT
    image += newline
    image ++= "255"
    image += newline
    for y in range!HEIGHT
        for x in range!WIDTH
            image = writePixel!(image x y WIDTH HEIGHT world)
        end
    end
    image = reverse!image
}

result = writeImage!world
}
